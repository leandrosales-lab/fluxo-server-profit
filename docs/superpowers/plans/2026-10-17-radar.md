# Radar de fluxo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ler todas as ações à vista ao vivo e listar cada iceberg do dia com giros, volume e desfecho (segurou / rompeu / virou sniper), numa tela "Radar".

**Architecture:** O detector de defesas ganha o evento `segurou` e um gancho `ao_evento`. `core/radar.py` mantém um `DefesaDetector` por papel e transforma os eventos em registros do dia. Uma conexão da DLL (processo filho) assina a lista de ações; cada negócio leva o papel. O `main.py` envolve `engine.add_tick`: todo negócio vai ao radar, só o do papel aberto vai ao `FlowEngine`. API `GET /radar`, `POST /operar/ativo`; tela `radar.html`.

**Tech Stack:** Python 3 stdlib, testes em `test_engine.py` (`ok/err/info`), HTML/JS ES modules, `node tools/test_telas.mjs`, `node tools/qa_telas.js` (Playwright).

**Spec:** `docs/superpowers/specs/2026-10-17-radar-design.md`

## Global Constraints

- Só ações à vista (bolsa B, tick R$ 0,01). Universo: SgmtNm `CASH`, MktNm `EQUITY-CASH`, SctyCtgyNm `SHARES` ou `UNIT`, ticker `[A-Z0-9]{4}(3|4|5|6|7|8|11)`, 1º termo de SpcfctnCd em `ON, PN, PNA, PNB, PNC, PNA*, PNB*, PNC*, UNT` (651 papéis em 16/10/2026).

```python
"""
RADAR DE FLUXO — icebergs de todas as ações
===========================================
Um DefesaDetector por papel (criado no primeiro negócio dele). Cada iceberg confirmado vira um
registro do dia, atualizado pelos eventos do detector: segurou (4 ticks a favor), rompida, virou
sniper. Spec: docs/superpowers/specs/2026-10-17-radar-design.md.
"""

from __future__ import annotations

from datetime import datetime, timedelta, timezone

from core import calibracao
from core.defesas import DefesaDetector

FUSO_B3 = timezone(timedelta(hours=-3))
LIMITE_LISTA = 300          # exibição; não é regra de mercado


def _dia(ts: float):
    return datetime.fromtimestamp(ts, FUSO_B3).date()


class Radar:
    def __init__(self, cfg, calibracao_tabela: dict | None = None):
        self.cfg = cfg
        self._tabela = calibracao.carregar() if calibracao_tabela is None else calibracao_tabela
        self.reset()

    def reset(self):
        self._detectores: dict = {}      # papel → DefesaDetector
        self._registros: dict = {}       # (papel, corretora, lado, nível) → registro
        self._vivos: dict = {}           # chave do registro → Defesa (enquanto o detector a acompanha)
        self._negocios = 0
        self._dia = None

    # ── entrada
    def on_tick(self, tick):
        dia = _dia(tick.timestamp)
        if self._dia is not None and dia != self._dia:
            self.reset()
        self._dia = dia
        ativo = tick.asset
        det = self._detectores.get(ativo)
        if det is None:
            det = DefesaDetector(self.cfg)
            det.ao_evento = lambda ev, d, ativo=ativo: self._ao_evento(ativo, ev, d)
            self._detectores[ativo] = det
        self._negocios += 1
        det.on_tick(tick)

    def envolver(self, add_tick, engine):
        """Entrega todo negócio ao radar e só o do papel aberto (engine.asset) ao motor."""
        def entregar(tick):
            if not tick.asset:
                tick.asset = engine.asset
            self.on_tick(tick)
            if tick.asset == engine.asset:
                add_tick(tick)
        return entregar

    # ── eventos do detector
    def _ao_evento(self, ativo, ev, d):
        chave = (ativo, d.corretora, d.lado, d.nivel)
        reg = self._registros.get(chave)
        if ev.evento == "confirmada" and d.tipo == "iceberg":
            if reg is None:
                reg = self._registros[chave] = {
                    "ativo": ativo, "corretora": d.corretora, "lado": d.lado, "preco": d.preco,
                    "giros": d.giros, "volume": d.vol_passivo, "confirmado_as": ev.ts,
                    "situacao": "segurando", "desfecho_as": None, "segurou_as": None, "rompeu_as": None,
                }
            self._vivos[chave] = d
            return
        if reg is None:
            return
        reg["giros"], reg["volume"] = d.giros, d.vol_passivo
        if ev.evento == "segurou":
            reg["segurou_as"] = ev.ts
            if reg["situacao"] == "segurando":
                reg["situacao"], reg["desfecho_as"] = "segurou", ev.ts
        elif ev.evento == "rompida":
            reg["rompeu_as"] = ev.ts
            if reg["situacao"] == "segurando":
                reg["situacao"], reg["desfecho_as"] = "rompeu", ev.ts
        elif ev.evento == "reclassificada" and d.tipo == "sniper":
            reg["situacao"], reg["desfecho_as"] = "virou sniper", ev.ts
            self._vivos.pop(chave, None)

    # ── consultas
    def _atualizar(self, chave, reg):
        d = self._vivos.get(chave)
        if d is not None:
            reg["giros"], reg["volume"] = d.giros, d.vol_passivo

    def lista(self, situacao: str | None = None, limite: int = LIMITE_LISTA) -> list[dict]:
        segurando, resolvidos = [], []
        for chave, reg in self._registros.items():
            self._atualizar(chave, reg)
            if situacao and reg["situacao"] != situacao:
                continue
            (segurando if reg["situacao"] == "segurando" else resolvidos).append(reg)
        segurando.sort(key=lambda r: -r["volume"])
        resolvidos.sort(key=lambda r: -(r["desfecho_as"] or 0))
        saida = []
        for reg in (segurando + resolvidos)[:limite]:
            r = dict(reg)
            r["acerto"] = calibracao.acerto_iceberg(self._tabela, r["ativo"], r["volume"])
            saida.append(r)
        return saida

    def resumo(self) -> dict:
        cont = {"segurando": 0, "segurou": 0, "rompeu": 0, "virou sniper": 0}
        for reg in self._registros.values():
            cont[reg["situacao"]] += 1
        return {"segurando": cont["segurando"], "seguraram": cont["segurou"], "romperam": cont["rompeu"],
                "viraram_sniper": cont["virou sniper"], "papeis": len(self._detectores),
                "negocios": self._negocios}
```

```python
"""
LISTA DAS AÇÕES À VISTA (cadastro público de instrumentos da B3)
================================================================
Baixa o InstrumentsConsolidatedFile do dia útil mais recente e grava config/acoes_b3.txt com as ações
do lote padrão: ON, PN e UNIT (sem fracionário, BDR, FII, ETF, recibo, direito). O radar assina essa lista.

Uso: python -X utf8 tools/acoes_b3.py [--data AAAA-MM-DD]
"""

import argparse
import csv
import json
import os
import re
import sys
import urllib.request
from datetime import date, timedelta

RAIZ = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SAIDA = os.path.join(RAIZ, "config", "acoes_b3.txt")
URL_PEDIDO = "https://arquivos.b3.com.br/api/download/requestname?fileName=InstrumentsConsolidatedFile&date={data}"
URL_ARQUIVO = "https://arquivos.b3.com.br/api/download/?token={token}"
TIPOS = {"ON", "PN", "PNA", "PNB", "PNC", "PNA*", "PNB*", "PNC*", "UNT"}
TICKER = re.compile(r"[A-Z0-9]{4}(3|4|5|6|7|8|11)")


def filtrar(linhas) -> list[str]:
    acoes = set()
    for x in csv.DictReader(linhas, delimiter=";"):
        espec = (x.get("SpcfctnCd") or "").split()
        if (x.get("SgmtNm") == "CASH" and x.get("MktNm") == "EQUITY-CASH"
                and x.get("SctyCtgyNm") in ("SHARES", "UNIT") and espec and espec[0] in TIPOS
                and TICKER.fullmatch(x.get("TckrSymb") or "")):
            acoes.add(x["TckrSymb"])
    return sorted(acoes)


def carregar_lista(caminho: str = SAIDA) -> list[str]:
    try:
        with open(caminho, encoding="utf-8") as f:
            return [l.strip().upper() for l in f if l.strip() and not l.lstrip().startswith("#")]
    except OSError:
        return []


def baixar(dia: date) -> list[str]:
    with urllib.request.urlopen(URL_PEDIDO.format(data=dia.isoformat()), timeout=30) as r:
        token = json.load(r)["redirectUrl"].split("token=")[1]
    with urllib.request.urlopen(URL_ARQUIVO.format(token=token), timeout=120) as r:
        texto = r.read().decode("latin-1")
    linhas = texto.splitlines()
    return linhas[1:] if linhas and linhas[0].startswith("Status") else linhas


def main():
    p = argparse.ArgumentParser(description="Atualiza config/acoes_b3.txt pelo cadastro da B3")
    p.add_argument("--data", help="Dia do cadastro (padrão: último dia útil)")
    a = p.parse_args()
    dia = date.fromisoformat(a.data) if a.data else date.today()
    for _ in range(7):                                  # volta até achar um dia com arquivo
        if dia.weekday() < 5:
            try:
                acoes = filtrar(baixar(dia))
                if acoes:
                    break
            except Exception as e:                      # noqa: BLE001 — tenta o dia anterior
                print(f"{dia}: {e}")
        dia -= timedelta(days=1)
    else:
        sys.exit("Não achei o cadastro da B3 nos últimos 7 dias")
    with open(SAIDA, "w", encoding="utf-8", newline="\n") as f:
        f.write(f"# Ações à vista (ON/PN/UNIT, lote padrão) — cadastro de instrumentos da B3 de {dia}\n")
        f.write("# Atualize com: python -X utf8 tools/acoes_b3.py\n")
        f.write("\n".join(acoes) + "\n")
    print(f"{len(acoes)} ações ({dia}) → {SAIDA}")


if __name__ == "__main__":
    main()
```

```js
import { atualizarLinhas } from "./tabela.js";

export function textoSituacao(r) {
  if (r.situacao === "segurando") return "segurando";
  if (r.situacao === "segurou") return `segurou às ${horaB3(r.segurou_as)}` + (r.rompeu_as ? `, rompeu às ${horaB3(r.rompeu_as)}` : "");

}

export function linhaRadar(r) {
  const classe = r.situacao === "segurando" ? "destaque" : "";
  return { classe, celulas: [
    { t: r.ativo }, { t: r.corretora }, { t: r.lado, c: r.lado },
    { t: preco(r.preco, 2), c: `num ${r.lado}` }, { t: inteiro(r.giros), c: "num" },

    { t: textoSituacao(r) }, { t: r.acerto ? pct(100 * r.acerto.taxa, 0) : "—", c: "num faint" }] };
}
```
